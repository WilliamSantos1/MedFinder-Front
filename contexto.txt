export const clinicSchema = clinicFields.superRefine((clinic, ctx) => {
  if (!clinic.isDemo && (!clinic.sourceUrl || !clinic.verifiedAt))
    ctx.addIssue({ code: 'custom', message: 'Clínicas reais exigem fonte e data de verificação.' });
  if ((clinic.latitude === null) !== (clinic.longitude === null))
    ctx.addIssue({ code: 'custom', message: 'Informe ambas as coordenadas ou nenhuma.' });
  if (
    !clinic.isDemo &&
    clinic.latitude !== null &&
    (!clinic.coordinatesVerifiedAt || !clinic.coordinatesSourceUrl)
  )
    ctx.addIssue({
      code: 'custom',
      message: 'Coordenadas reais exigem fonte e verificação próprias.',
    });
  if (
    [clinic.verifiedAt, clinic.coordinatesVerifiedAt].some(
      (date) => date && Date.parse(date) > Date.now(),
    )
  )
    ctx.addIssue({ code: 'custom', message: 'A verificação não pode estar no futuro.' });
});

function allowedWebOrigins(config: Config) {
  const configured = new URL(config.WEB_ORIGIN);
  if (
    !['http:', 'https:'].includes(configured.protocol) ||
    configured.username ||
    configured.password ||
    configured.pathname !== '/' ||
    configured.search ||
    configured.hash
  )
    throw new Error(
      'WEB_ORIGIN deve conter somente protocolo, endereço e porta, por exemplo http://localhost:5173.',
    );

  // Browsers serialize Origin without a trailing slash. Keep protocol and port
  // exact, accepting loopback aliases only in local development/test.
  const origins = new Set([configured.origin]);
  const loopbackHosts = ['localhost', '127.0.0.1', '[::1]'];
  if (config.NODE_ENV !== 'production' && loopbackHosts.includes(configured.hostname)) {
    for (const hostname of loopbackHosts) {
      const alias = new URL(configured.origin);
      alias.hostname = hostname;
      origins.add(alias.origin);
    }
  }
  return origins;
}

export async function buildApp(
  config: Config,
  db: Database,
  provider: AIProvider = createProvider(config),
) {
  const webOrigins = allowedWebOrigins(config);
  const app = Fastify({
    logger: config.NODE_ENV === 'test' ? false : { level: 'info' },
    logController: new LogController({ disableRequestLogging: true }),
    bodyLimit: 24000,
    requestTimeout: 60000,
    trustProxy: config.TRUST_PROXY_HOPS ? (_address, hop) => hop < config.TRUST_PROXY_HOPS : false,
  });
  await app.register(helmet, {
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'self'"],
        scriptSrc: ["'self'"],
        styleSrc: ["'self'"],
        imgSrc: ["'self'", 'data:'],
        connectSrc: ["'self'"],
        objectSrc: ["'none'"],
        frameAncestors: ["'none'"],
        upgradeInsecureRequests: config.NODE_ENV === 'production' ? [] : null,
      },
    },
  });
  await app.register(cors, {
    origin: [...webOrigins],
    methods: ['GET', 'POST'],
    credentials: false,
  });
  await app.register(rateLimit, { max: config.HTTP_RATE_LIMIT, timeWindow: '1 minute' });
  const catalog = new CatalogRepository(db, config.ALLOW_DEMO_DATA);
  const feedback = new FeedbackRepository(db);
  const chat = new ChatService(
    new Retriever(db, provider, config.ALLOW_DEMO_DATA),
    provider,
    catalog,
    new SpecialtyResolver(new SpecialtyRepository(db)),
    config.AI_MAX_CONCURRENCY,
    () => app.log.warn({ event: 'ai_unavailable' }, 'Consulta de IA indisponível'),
  );

  app.addHook('onRequest', async (request, reply) => {
    if (request.url.startsWith('/api/')) reply.header('Cache-Control', 'no-store');
    if (
      request.method === 'POST' &&
      request.headers.origin &&
      !webOrigins.has(request.headers.origin)
    ) {
      return reply.code(403).send({
        error:
          config.NODE_ENV === 'production'
            ? 'Origem não permitida.'
            : `Origem não permitida. Abra o MedFinder em ${new URL(config.WEB_ORIGIN).origin}.`,
      });
    }
  });
  app.get('/api/health', async () => ({ status: 'ok' }));
  app.get('/api/ready', async (_request, reply) => {
    try {
      await db.query('SELECT 1');
      const status = await catalogStatus(db, config, provider);
      const ready = status.directory.total > 0 && status.specialties.length > 0;
      return reply
        .code(ready ? 200 : 503)
        .send({ status: ready ? 'ready' : 'catalog_unavailable' });
    } catch {
      return reply.code(503).send({ status: 'unavailable' });
    }
  });
  app.get('/api/catalog', async () => catalogStatus(db, config, provider));
  app.get('/api/clinics', async (request) =>
    catalog.search(clinicFiltersSchema.parse(request.query)),
  );
  app.post('/api/clinics/search', async (request) =>
    catalog.search(clinicSearchSchema.parse(request.body)),
  );
  app.post(
    '/api/feedback',
    { config: { rateLimit: { max: config.FEEDBACK_RATE_LIMIT, timeWindow: '1 minute' } } },
    async (request, reply) => {
      const saved = await feedback.save(feedbackRequestSchema.parse(request.body));
      return saved
        ? { status: 'saved' }
        : reply
            .code(400)
            .send({ error: 'Feedback inválido ou expirado. Confira o motivo e tente novamente.' });
    },
  );
  app.post(
    '/api/chat',
    { config: { rateLimit: { max: config.CHAT_RATE_LIMIT, timeWindow: '1 minute' } } },
    async (request) => {
      const body = chatRequestSchema.parse(request.body);
      const response = await chat.answer(body);
      try {
        response.feedback = await feedback.issue(
          response,
          Boolean(response.context?.insurance || body.insurance),
        );
      } catch {
        app.log.warn(
          { event: 'feedback_receipt_unavailable' },
          'Feedback temporariamente indisponível',
        );
      }
      return response;
    },
  );
  app.setErrorHandler((error, _request, reply) => {
    if (error instanceof ZodError)
      return reply.code(400).send({
        error: 'Revise os campos enviados e seus limites. A solicitação contém dados inválidos.',
      });
    const statusCode =
      error instanceof Error && 'statusCode' in error && typeof error.statusCode === 'number'
        ? error.statusCode
        : 500;
    if (statusCode < 500) {
      const message =
        statusCode === 429
          ? 'Muitas solicitações. Aguarde um minuto e tente novamente.'
          : 'Não foi possível processar a solicitação.';
      return reply.code(statusCode).send({ error: message });
    }
    app.log.error({ event: 'request_failed' }, 'Falha interna na solicitação');
    return reply.code(500).send({ error: 'Serviço indisponível. Tente novamente em instantes.' });
  });

  const dist = resolve('dist');
  if (existsSync(resolve(dist, 'index.html'))) {
    await app.register(fastifyStatic, { root: dist, wildcard: false });
    app.setNotFoundHandler((request, reply) => {
      if (request.url.startsWith('/api/') || request.url.includes('.') || request.method !== 'GET')
        return reply.code(404).send({ error: 'Recurso não encontrado.' });
      return reply.header('Cache-Control', 'no-cache').sendFile('index.html');
    });
  } else
    app.setNotFoundHandler((_request, reply) =>
      reply.code(404).send({ error: 'Recurso não encontrado.' }),
    );
  return app;
}

=================== server/index.ts ===================
import { buildApp } from './app.js';
import { createProvider } from './ai/factory.js';
import { readConfig } from './config.js';
import { connectDatabase } from './db/database.js';
import { migrate } from './db/migrations.js';
import { seedDevelopment } from './db/seed.js';

async function main() {
  const config = readConfig();
  const db = await connectDatabase(config);
  try {
    if (config.NODE_ENV !== 'production') await migrate(db);
    const provider = createProvider(config);
    if (config.AUTO_SEED && config.ALLOW_DEMO_DATA) {
      await seedDevelopment(db, provider);
    }
    const app = await buildApp(config, db, provider);
    app.addHook('onClose', async () => {
      await db.close();
    });
    await app.listen({ port: config.PORT, host: config.HOST });
    for (const signal of ['SIGINT', 'SIGTERM'])
      process.once(signal, () => {
        void app.close().catch(() => {
          process.exitCode = 1;
        });
      });
  } catch (error) {
    await db.close();
    throw error;
  }
}

main().catch(() => {
  console.error(
    'Não foi possível iniciar o MedFinder. Verifique o .env, a conexão e as migrações.',
  );
  process.exitCode = 1;
});

  CHAT_RATE_LIMIT: z.coerce.number().int().min(1).max(120).default(12),
  HTTP_RATE_LIMIT: z.coerce.number().int().min(1).max(1000).default(120),
  FEEDBACK_RATE_LIMIT: z.coerce.number().int().min(1).max(120).default(20),
  AI_MAX_CONCURRENCY: z.coerce.number().int().min(1).max(50).default(4),
});

export function readConfig(env: NodeJS.ProcessEnv = process.env) {
  const config = schema.parse(env);
  if (config.AI_PROVIDER === 'openai' && !config.OPENAI_API_KEY) {
    throw new Error('AI_PROVIDER=openai exige OPENAI_API_KEY no servidor.');
  }
  if (config.AI_PROVIDER === 'ollama') {
    const url = new URL(config.OLLAMA_BASE_URL);
    if (
      url.protocol !== 'http:' ||
      !['127.0.0.1', 'localhost', '[::1]'].includes(url.hostname) ||
      url.username ||
      url.password ||
      url.pathname !== '/' ||
      url.search ||
      url.hash
    )
      throw new Error(
        'OLLAMA_BASE_URL deve apontar para o Ollama local, por exemplo http://127.0.0.1:11434.',
      );
    for (const model of [config.OLLAMA_MODEL, config.OLLAMA_EMBEDDING_MODEL]) {
      if (!/^[a-z0-9][a-z0-9._:/-]*$/i.test(model) || /cloud/i.test(model))
        throw new Error('Use modelos Ollama baixados localmente, sem sufixo cloud.');
    }
  }
  if (config.NODE_ENV === 'production') {
    if (
      !config.DATABASE_URL ||
      config.ALLOW_DEMO_DATA ||
      config.AUTO_SEED ||
      config.AI_PROVIDER === 'demo'
    ) {
      throw new Error(
        'Produção exige PostgreSQL externo, IA generativa, ALLOW_DEMO_DATA=false e AUTO_SEED=false.',
      );
    }
    if (new URL(config.WEB_ORIGIN).protocol !== 'https:')
      throw new Error('WEB_ORIGIN deve usar HTTPS em produção.');
  }
  return config;
}
export type Config = ReturnType<typeof readConfig>;

import { PGlite } from '@electric-sql/pglite';
import { vector } from '@electric-sql/pglite-pgvector';
import pg from 'pg';
import { mkdir } from 'node:fs/promises';
import { dirname, resolve } from 'node:path';
import type { Config } from '../config.js';

export interface SqlClient {
  query<T>(sql: string, params?: unknown[]): Promise<{ rows: T[] }>;
  exec(sql: string): Promise<void>;
}
export interface Database extends SqlClient {
  transaction<T>(run: (tx: SqlClient) => Promise<T>): Promise<T>;
  close(): Promise<void>;
}

export async function connectDatabase(config: Config): Promise<Database> {
  if (!config.DATABASE_URL) {
    if (config.DATA_DIR !== 'memory://')
      await mkdir(dirname(resolve(config.DATA_DIR)), { recursive: true });
    const db = new PGlite(config.DATA_DIR, { extensions: { vector } });
    await db.waitReady;
    return {
      query: (sql, params) => db.query(sql, params),
      exec: async (sql) => {
        await db.exec(sql);
      },
      transaction: (run) =>
        db.transaction((tx) =>
          run({
            query: (sql, params) => tx.query(sql, params),
            exec: async (sql) => {
              await tx.exec(sql);
            },
          }),
        ),
      close: () => db.close(),
    };
  }
  const pool = new pg.Pool({
    connectionString: config.DATABASE_URL,
    max: 10,
    connectionTimeoutMillis: 10000,
    statement_timeout: 10000,
    ssl: config.DATABASE_SSL ? { rejectUnauthorized: true } : undefined,
  });
  await pool.query('SELECT 1');
  return {
    query: async <T>(sql: string, params?: unknown[]) => ({
      rows: (await pool.query(sql, params)).rows as T[],
    }),
    exec: async (sql) => {
      await pool.query(sql);
    },
    transaction: async (run) => {
      const client = await pool.connect();
      try {
        await client.query('BEGIN');
        const value = await run({
          query: async <T>(sql: string, params?: unknown[]) => ({
            rows: (await client.query(sql, params)).rows as T[],
          }),
          exec: async (sql) => {
            await client.query(sql);
          },
        });
        await client.query('COMMIT');
        return value;
      } catch (error) {
        await client.query('ROLLBACK');
        throw error;
      } finally {
        client.release();
      }
    },
    close: () => pool.end(),
  };
}

export async function migrate(db: Database) {
  // Only an operator runs migrations in production; one startup process locally.
  await db.exec(
    'CREATE TABLE IF NOT EXISTS schema_migrations (version integer PRIMARY KEY, applied_at timestamptz NOT NULL DEFAULT now())',
  );
  for (const migration of [...migrations, ...evolutionMigrations]) {
    await db.transaction(async (tx) => {
      await tx.exec('LOCK TABLE schema_migrations IN EXCLUSIVE MODE');
      const { rows } = await tx.query('SELECT version FROM schema_migrations WHERE version=$1', [
        migration.version,
      ]);
      if (rows.length) return;
      await tx.exec(migration.sql);
      await tx.query('INSERT INTO schema_migrations(version) VALUES ($1)', [migration.version]);
    });
  }
}

import { createHash, randomBytes, randomUUID } from 'node:crypto';
import type { ChatResponse, FeedbackReason, FeedbackRequest } from '../../shared/contracts.js';
import { PROMPT_VERSION } from '../ai/prompt.js';
import { redactIdentifiers } from '../domain/text.js';
import type { Database } from './database.js';

const digest = (token: string) => createHash('sha256').update(token).digest('hex');
export class FeedbackRepository {
  constructor(private readonly db: Database) {}

  async issue(
    response: ChatResponse,
    insuranceUsed: boolean,
  ): Promise<NonNullable<ChatResponse['feedback']>> {
    const reasons: FeedbackReason[] = ['unclear_answer', 'not_helpful', 'too_slow', 'other'];
    if (response.specialty) reasons.unshift('incorrect_specialty');
    if (response.clinics.length) reasons.push('clinic_problem');
    if (response.locationUsed) reasons.push('location_problem');
    if (insuranceUsed && (response.specialty || response.intent !== 'care_navigation'))
      reasons.push('insurance_problem');
    const token = randomBytes(32).toString('hex');
    await this.db.query(
      `INSERT INTO response_receipts(response_id,token_hash,mode,prompt_version,allowed_reasons)
      VALUES($1,$2,$3,$4,$5)`,
      [response.id, digest(token), response.mode, PROMPT_VERSION, reasons],
    );
    return { token, reasons };
  }

  async save(input: FeedbackRequest): Promise<boolean> {
    // The receipt is a capability: clients cannot forge metadata or vote on arbitrary response IDs.
    // Only metadata is stored; there is no chat body, location, prompt or answer here.
    const comment =
      input.rating === 'negative' && input.comment
        ? redactIdentifiers(input.comment).slice(0, 500)
        : null;
    const result = await this.db.query(
      `
      INSERT INTO ai_feedback(id,response_id,rating,reason,comment,mode,prompt_version)
      SELECT $1,r.response_id,$2,$3,$4,r.mode,r.prompt_version FROM response_receipts r
      WHERE r.response_id=$5 AND r.token_hash=$6 AND r.created_at>now()-interval '7 days'
        AND ($2='positive' OR $3=ANY(r.allowed_reasons))
      ON CONFLICT(response_id) DO UPDATE SET rating=EXCLUDED.rating,reason=EXCLUDED.reason,
        comment=EXCLUDED.comment,updated_at=now()
      RETURNING id`,
      [
        randomUUID(),
        input.rating,
        input.reason ?? null,
        comment,
        input.responseId,
        digest(input.token),
      ],
    );
    return result.rows.length === 1;
  }

  async purgeExpiredReceipts() {
    await this.db.query("DELETE FROM response_receipts WHERE created_at<=now()-interval '7 days'");
  }
}

import {
  clinicSchema,
  type Clinic,
  type ClinicFilters,
  type ClinicResults,
  type Specialty,
} from '../../shared/contracts.js';
import { normalize } from '../domain/text.js';
import { distanceKm } from '../domain/distance.js';
import type { Database, SqlClient } from './database.js';
import { SpecialtyRepository } from './specialties.js';

export class CatalogRepository {
  constructor(
    private readonly db: Database,
    private readonly allowDemo: boolean,
  ) {}

  async import(clinics: Clinic[]) {
    const validated = clinics.map((clinic) => clinicSchema.parse(clinic));
    if (new Set(validated.map((clinic) => clinic.id)).size !== validated.length)
      throw new Error('O catálogo contém IDs duplicados.');
    if (!this.allowDemo && validated.some((clinic) => clinic.isDemo))
      throw new Error('Clínicas fictícias não são permitidas neste ambiente.');
    await this.db.transaction((tx) => CatalogRepository.write(tx, validated));
  }

  static async write(tx: SqlClient, validated: Clinic[]) {
    const requested = [...new Set(validated.flatMap((c) => c.specialties))];
    const registered = await tx.query<{ id: string }>(
      'SELECT id FROM specialties WHERE active AND id=ANY($1::text[]) FOR SHARE',
      [requested],
    );
    if (requested.some((id) => !registered.rows.some((row) => row.id === id)))
      throw new Error('Cadastre e ative as especialidades antes de importar as clínicas.');
    for (const clinic of validated) {
      await tx.query(
        `INSERT INTO clinics(id,city_key,insurance_keys,data,latitude,longitude,coordinates_verified_at,coordinates_source_url)
          VALUES($1,$2,$3,$4::jsonb,$5,$6,$7,$8)
          ON CONFLICT(id) DO UPDATE SET city_key=$2,insurance_keys=$3,data=$4::jsonb,
          latitude=$5,longitude=$6,coordinates_verified_at=$7,coordinates_source_url=$8`,
        [
          clinic.id,
          normalize(clinic.city),
          clinic.insurances.map(normalize),
          JSON.stringify(clinic),
          clinic.latitude,
          clinic.longitude,
          clinic.coordinatesVerifiedAt,
          clinic.coordinatesSourceUrl,
        ],
      );
      await tx.query('DELETE FROM clinic_specialties WHERE clinic_id=$1', [clinic.id]);
      await tx.query(
        `INSERT INTO clinic_specialties(clinic_id,specialty_id)
        SELECT $1,unnest($2::text[]) ON CONFLICT DO NOTHING`,
        [clinic.id, clinic.specialties],
      );
    }
  }

  private visibility = `(data->>'active')::boolean = true AND
    (($1::boolean AND (data->>'isDemo')::boolean) OR
      ((data->>'isDemo')::boolean = false AND (data->>'verifiedAt')::timestamptz >= now() -
        interval '180 days' AND (data->>'verifiedAt')::timestamptz <= now()))`;

  async search(
    filters: ClinicFilters,
    specialtyIds?: string[],
    clinicIds?: string[],
  ): Promise<ClinicResults> {
    const ids = specialtyIds ?? (filters.specialty ? [filters.specialty] : []);
    const { rows } = await this.db.query<{
      data: Clinic;
      latitude: number | null;
      longitude: number | null;
      details: Specialty[];
    }>(
      `SELECT c.data,c.latitude,c.longitude,
        (SELECT COALESCE(jsonb_agg(jsonb_build_object('id',s.id,'name',s.name) ORDER BY s.name,s.id),'[]')
          FROM clinic_specialties cs JOIN specialties s ON s.id=cs.specialty_id
          WHERE cs.clinic_id=c.id AND s.active) AS details
        FROM clinics c WHERE ${this.visibility}
          AND ($6::text[] IS NULL OR c.id=ANY($6::text[]))
          AND ($2='' OR city_key=$2) AND ($3='' OR $3=ANY(insurance_keys))
          AND EXISTS (SELECT 1 FROM clinic_specialties cs JOIN specialties s ON s.id=cs.specialty_id
            WHERE cs.clinic_id=c.id AND s.active AND (cardinality($4::text[])=0 OR s.id=ANY($4::text[])))
          AND NOT EXISTS (SELECT 1 FROM unnest($5::text[]) AS term
            WHERE strpos(translate(lower(concat_ws(' ',data->>'name',data->>'neighborhood',data->>'address')),
              'áàâãäéèêëíìîïóòôõöúùûüç', 'aaaaaeeeeiiiiooooouuuuc'),term)=0)`,
      [
        this.allowDemo,
        normalize(filters.city),
        normalize(filters.insurance),
        ids,
        normalize(filters.q ?? '')
          .split(/\s+/)
          .filter(Boolean),
        clinicIds ?? null,
      ],
    );
    // Small curated MVP catalog: filter in SQL, compute/sort ALL candidates, then paginate.
    const items = rows.map(({ data, latitude, longitude, details }) => ({
      ...data,
      latitude,
      longitude,
      specialties: details.map((s) => s.id),
      specialtyDetails: details,
      distanceKm:
        filters.location && latitude !== null && longitude !== null
          ? distanceKm(filters.location.latitude, filters.location.longitude, latitude, longitude)
          : null,
    }));
    items.sort((a, b) => {
      if (filters.location) {
        const proximity = (a.distanceKm ?? Infinity) - (b.distanceKm ?? Infinity);
        if (proximity && !Number.isNaN(proximity)) return proximity;
      } else if (a.isDemo !== b.isDemo) return Number(a.isDemo) - Number(b.isDemo);
      return a.name.localeCompare(b.name, 'pt-BR') || a.id.localeCompare(b.id);
    });
    return {
      items: items.slice((filters.page - 1) * 6, filters.page * 6),
      total: items.length,
      page: filters.page,
      pageSize: 6,
      order: filters.location ? 'proximity' : 'name',
    };
  }

  async mentionedClinics(text: string): Promise<string[]> {
    const words = new Set(
      normalize(text)
        .replace(/[^a-z0-9]+/g, ' ')
        .split(/\s+/),
    );
    const generic = new Set([
      'clinica',
      'clinicas',
      'medica',
      'medico',
      'centro',
      'instituto',
      'fortaleza',
      'eusebio',
      'maracanau',
      'unidade',
      'de',
      'do',
      'da',
      'dos',
      'das',
      'saude',
    ]);
    const rows = (
      await this.db.query<{ id: string; name: string }>(
        `SELECT id,data->>'name' AS name FROM clinics WHERE ${this.visibility}`,
        [this.allowDemo],
      )
    ).rows;
    const taxonomy = await this.db.query<{ normalized_name: string; aliases: string[] }>(
      'SELECT normalized_name,aliases FROM specialties',
    );
    for (const entry of taxonomy.rows)
      for (const term of [entry.normalized_name, ...entry.aliases])
        for (const word of term.split(/\s+/)) generic.add(word);
    const matches = rows.map((row) => ({
      id: row.id,
      score: normalize(row.name)
        .replace(/[^a-z0-9]+/g, ' ')
        .split(/\s+/)
        .filter((word) => word.length >= 3 && !generic.has(word) && words.has(word)).length,
    }));
    const maximum = Math.max(0, ...matches.map((match) => match.score));
    return maximum
      ? matches.filter((match) => match.score === maximum).map((match) => match.id)
      : [];
  }

  async metadata() {
    const { rows } = await this.db.query<{ data: Clinic }>(
      `SELECT data FROM clinics c WHERE ${this.visibility}
        AND EXISTS (SELECT 1 FROM clinic_specialties cs JOIN specialties s ON s.id=cs.specialty_id
          WHERE cs.clinic_id=c.id AND s.active)`,
      [this.allowDemo],
    );
    return {
      cities: [...new Set(rows.map(({ data }) => data.city))].sort(),
      insurances: [...new Set(rows.flatMap(({ data }) => data.insurances))].sort(),
      specialties: await new SpecialtyRepository(this.db).list(),
      demoData: rows.some(({ data }) => data.isDemo),
      total: rows.length,
      realRecords: rows.filter(({ data }) => !data.isDemo).length,
      sourceCount: new Set(
        rows
          .filter(({ data }) => !data.isDemo && data.sourceUrl)
          .map(({ data }) => new URL(data.sourceUrl!).hostname.replace(/^www\./, '')),
      ).size,
      lastVerifiedAt:
        rows
          .filter(({ data }) => !data.isDemo && data.verifiedAt)
          .map(({ data }) => data.verifiedAt!)
          .sort()
          .at(-1) ?? null,
    };
  }
}

import { randomUUID } from 'node:crypto';
import { ZodError } from 'zod';
import {
  modelAnswerSchema,
  type ChatRequest,
  type ChatResponse,
  type Specialty,

export function safetyResponse(urgency: 'emergency' | 'prompt'): ChatResponse {
  return {
    ...empty(),
    urgency,
    mode: 'safety',
    status: 'ok',
    answer:

function fallback(status: 'llm_unavailable' | 'invalid_output'): ChatResponse {
  return {
    ...empty(),
    urgency: 'uncertain',
    mode: 'fallback',
    status,
    answer:
      status === 'llm_unavailable'
        ? 'A IA está indisponível neste momento. Tente novamente em instantes. O catálogo continua disponível para consulta. Em caso de urgência, procure atendimento presencial sem esperar pelo chat.'
        : 'Não foi possível validar a orientação gerada. Tente descrever o desconforto de outra forma ou procure avaliação presencial. Em caso de urgência, não espere pelo chat.',

}
export class ChatService {
  private inFlight = 0;
  constructor(
    private readonly retriever: Retriever,
    private readonly provider: AIProvider,
    private readonly catalog: CatalogRepository,
    private readonly resolver: SpecialtyResolver,
    private readonly maxConcurrency = 4,
    private readonly reportFailure: () => void = () => {},
  ) {}

  async answer(request: ChatRequest): Promise<ChatResponse> {
    const urgency = detectUrgency([...request.history, request.message]);
    if (urgency) return safetyResponse(urgency);
    if (earObjectConcern([...request.history, request.message])) {
      const explanation = careExplanation([...request.history, request.message]);
      const specialty = await this.resolver.byId('otorrinolaringologia');
      return {
        ...safetyResponse('prompt'),
        answer: `${explanation.text} Procure avaliação presencial ainda hoje, sem esperar uma consulta eletiva. Não introduza outros objetos no ouvido.`,
        specialty: specialty
          ? {
              ...specialty,
              reason:
                'Área relacionada à avaliação do ouvido. A prioridade é o atendimento presencial hoje.',
            }
          : null,
        sources: explanation.sources,
      };
    }
    const route = await routeCatalog(request, this.catalog, this.resolver);
    if (route) return this.catalogAnswer(request, route);
    const conversation = careConversation(request);
    const assessment = assessDetail([...conversation.history, request.message]);
    const patient = patientProfile(
      [...conversation.history, request.message],
      conversation.care?.questions.includes('age_group'),
    );
    if (patient.ageYears !== null || patient.adultReported) assessment.known.add('age_group');
    if (this.provider.mode === 'openai' && !request.consent)
      throw Object.assign(new Error('Autorize o envio à OpenAI antes de continuar.'), {
        statusCode: 400,
      });
    if (this.inFlight >= this.maxConcurrency) return fallback('llm_unavailable');
    this.inFlight++;
    try {
      let output: unknown;
      try {
        // No coordinates, client context, filters or catalog facts enter the LLM.
        output = await this.provider.generate({
          message: redactIdentifiers(request.message),
          history: conversation.history.map(redactIdentifiers),
          specialties: await this.resolver.list(),
          previousQuestions: (conversation.care?.questions ?? []).map((topic) =>
            questionText(topic, assessDetail(conversation.history).text),
          ),
          patient,
        });
      } catch (error) {
        this.reportFailure();
        return fallback(
          error instanceof ZodError || error instanceof SyntaxError
            ? 'invalid_output'
            : 'llm_unavailable',
        );
      }
      const parsed = modelAnswerSchema.safeParse(output);
      if (!parsed.success) return fallback('invalid_output');
      const draft = parsed.data;
      if (draft.urgency === 'emergency' || draft.urgency === 'prompt')
        return safetyResponse(draft.urgency);
      const generated = [
        draft.answer,
        ...draft.questions,
        draft.specialty?.name,
        draft.specialty?.reason,
      ]
        .filter(Boolean)
        .join(' ');
      if (isUnsafeOutput(generated) || containsCatalogClaim(generated))
        return fallback('invalid_output');
      const base: ChatResponse = {
        ...empty(),
        status: 'needs_information',
        urgency: 'uncertain',
        mode: this.provider.mode,
        answer:
          'Ainda não consegui determinar uma especialidade com segurança a partir desse relato.',
      };
      const needsAge =
        patient.needsAge ||
        (/^pediatr/i.test(draft.specialty?.name ?? '') && !assessment.known.has('age_group'));
      if (assessment.generic || needsAge || !draft.specialty) {
        const relevant = assessment.generic
          ? draft.questions.filter(
              (topic) => topic !== 'age_group' || !assessment.known.has('age_group'),
            )
          : [];
        const topics =
          assessment.generic || needsAge
            ? clarificationQuestions(
                {
                  ...assessment,
                  missing: needsAge ? ['age_group', ...assessment.missing] : assessment.missing,
                },
                relevant,
                conversation.care,
              )
            : [];
        return {
          ...base,
          answer: topics.length
            ? needsAge
              ? 'A idade de quem está com o sintoma pode mudar a indicação de atendimento. Preciso confirmar essa informação antes de recomendar uma especialidade.'
              : 'Antes de sugerir uma especialidade, preciso esclarecer alguns detalhes sobre o desconforto.'
            : 'Não consegui validar uma recomendação de especialidade a partir desse relato. Procure avaliação presencial para definir o atendimento adequado; não espere pelo chat se houver piora ou sinais de urgência.',
          questions: topics.map((topic) => questionText(topic, assessment.text)),
          context: {
            specialtyId: null,
            clinicIds: [],
            city: request.city,
            insurance: request.insurance,
            care: {
              questions: topics,
              askedTopics: [...new Set([...(conversation.care?.askedTopics ?? []), ...topics])],
              round: Math.min(3, (conversation.care?.round ?? 0) + 1),
            },
          },
        };
      }
      const match = await this.resolver.resolve(draft.specialty.name);
      if (match.status !== 'resolved' || !match.specialty)
        return {
          ...base,
          status:
            match.status === 'not_registered'
              ? 'specialty_not_registered'
              : 'specialty_unrecognized',
        };
      if (match.specialty.id === 'pediatria' && patient.adultReported)
        return fallback('invalid_output');
      // Model prose is never published: it can contain aliases, additional specialties or invented names.
      // The LLM supplies the candidate and urgency; presentation uses the resolved entity only.
      const explanation = careExplanation(
        [...conversation.history, request.message],
        match.specialty,
      );
      const response: ChatResponse = {
        ...base,
        questions: [],
        status: 'ok',
        urgency: draft.urgency,
        answer: `${this.provider.mode === 'demo' ? 'Exemplo demonstrativo, sem avaliação clínica. ' : ''}${explanation.text}`,
        sources: explanation.sources,
        specialty: {
          ...match.specialty,
          reason:
            match.specialty.id === 'pediatria'
              ? 'Área sugerida para avaliação inicial considerando a idade informada e o relato. Confirme a faixa etária atendida pela unidade.'
              : 'Área sugerida para avaliação do relato, sem estabelecer um diagnóstico.',
        },
      };
      return this.findClinics(request, response, match.specialty, request.city, request.insurance);
    } finally {
      this.inFlight--;
    }
  }

  private async catalogAnswer(request: ChatRequest, route: CatalogIntent): Promise<ChatResponse> {
    const base: ChatResponse = {
      ...empty(),
      intent: route.intent,
      urgency: 'routine',
      mode: 'catalog',
      status: 'ok',
      specialty: route.specialty,
      answer:
        'Consulte os dados publicados nas fontes de cada unidade. Confirme cobertura e disponibilidade diretamente com a clínica.',
    };
    if (route.clarification)
      return { ...base, status: 'needs_information', specialty: null, answer: route.clarification };
    const response = await this.findClinics(
      request,
      base,
      route.specialty,
      route.city,
      route.insurance,
      route.clinicIds,
    );
    if (route.nearest && !request.location)
      response.questions = [
        'Ative “Usar minha localização” e repita a pergunta para ordenar por distância.',
      ];
    if (route.intent === 'clinic_information' && response.clinics.length) {
      response.answer = route.detailsUnavailable
        ? 'Horários, preços e vagas não são confirmados pelo MedFinder. Consulte o contato e os textos da fonte oficial da unidade abaixo.'
        : response.clinics
            .map(
              (clinic) =>
                `${clinic.name}: ${clinic.address}, ${clinic.neighborhood ? clinic.neighborhood + ', ' : ''}${clinic.city}${clinic.state ? ' — ' + clinic.state : ''}.`,
            )
            .join('\n');
    }
    return response;
  }

  private async findClinics(
    request: ChatRequest,
    base: ChatResponse,
    specialty: Specialty | null,
    city: string,
    insurance: string,
    clinicIds?: string[],
  ): Promise<ChatResponse> {
    const results = await this.catalog.search(
      { city, insurance, specialty: specialty?.id ?? '', page: 1, location: request.location },
      undefined,
      clinicIds,
    );
    const clinics = results.items.slice(0, 3);
    const response: ChatResponse = {
      ...base,
      clinics,
      locationUsed: Boolean(request.location),
      status: clinics.length ? 'ok' : 'no_matching_clinics',
      context: {
        specialtyId: specialty?.id ?? null,
        clinicIds: clinics.map((c) => c.id),
        city,
        insurance,
      },
    };
    if (!clinics.length || (this.provider.mode === 'openai' && !request.consent)) return response;
    try {
      // Enrichment is restricted to selected IDs; geolocation stays out of the query.
      const rag = await this.retriever.search(
        [redactIdentifiers(request.message), specialty?.name].filter(Boolean).join('\n'),
        clinics.map((c) => c.id),
      );
      return { ...response, sources: [...response.sources, ...rag.sources], ragStatus: rag.status };
    } catch {
      return { ...response, ragStatus: 'unavailable' };
    }
  }
}

import {
  chatResponseSchema,
  type CatalogMeta,
  type ChatRequest,
  clinicResultsSchema,
  type FeedbackRequest,
} from '../../shared/contracts';

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const response = await fetch(`/api${path}`, {
    ...options,
    headers: { 'Content-Type': 'application/json', ...options.headers },
  });
  if (!response.ok) {
    const body = (await response.json().catch(() => ({}))) as { error?: string };
    throw new Error(body.error ?? 'Não foi possível conectar. Tente novamente.');
  }
  return response.json() as Promise<T>;
}

export const api = {
  catalog: (signal: AbortSignal) => request<CatalogMeta>('/catalog', { signal }),
  clinics: async (body: string, signal: AbortSignal) =>
    clinicResultsSchema.parse(await request('/clinics/search', { method: 'POST', body, signal })),
  feedback: (body: FeedbackRequest) =>
    request<{ status: 'saved' }>('/feedback', {
      method: 'POST',
      body: JSON.stringify(body),
      signal: AbortSignal.timeout(15000),
    }),
  chat: async (body: ChatRequest, signal: AbortSignal) =>
    chatResponseSchema.parse(
      await request('/chat', { method: 'POST', body: JSON.stringify(body), signal }),
    ),
};

import { useEffect, useRef, useState } from 'react';
import type { ChatRequest, ChatResponse } from '../../../shared/contracts';
import { api } from '../../lib/api';

export type Message =
  | { id: string; role: 'user'; text: string }
  | { id: string; role: 'assistant'; response: ChatResponse };

export function useChat(timeoutMs = 60000) {
  const [messages, setMessages] = useState<Message[]>([]);
  const [pending, setPending] = useState(false);
  const [error, setError] = useState('');
  const lastFilters = useRef<{ city: string; insurance: string } | null>(null);
  const controller = useRef<AbortController | null>(null);
  useEffect(() => () => controller.current?.abort(), []);

  async function send(input: Omit<ChatRequest, 'history' | 'context'>) {
    if (controller.current) return false;
    const current = new AbortController();
    controller.current = current;
    const userMessage: Message = { id: crypto.randomUUID(), role: 'user', text: input.message };
    const history = messages
      .filter((item) => item.role === 'user')
      .map((item) => item.text)
      .slice(-6);
    setMessages((previous) => [...previous, userMessage]);
    setError('');
    setPending(true);
    const timeout = window.setTimeout(() => current.abort(), timeoutMs);
    try {
      const previous = [...messages].reverse().find((item) => item.role === 'assistant');
      const context =
        previous?.role === 'assistant' && previous.response.context
          ? {
              ...previous.response.context,
              city:
                lastFilters.current?.city === input.city
                  ? previous.response.context.city
                  : input.city,
              insurance:
                lastFilters.current?.insurance === input.insurance
                  ? previous.response.context.insurance
                  : input.insurance,
            }
          : undefined;
      const response = await api.chat({ ...input, history, context }, current.signal);
      lastFilters.current = { city: input.city, insurance: input.insurance };
      setMessages((previous) => [...previous, { id: response.id, role: 'assistant', response }]);
      return true;
    } catch (failure) {
      setMessages((previous) => previous.filter((item) => item.id !== userMessage.id));
      setError(
        current.signal.aborted
          ? 'Envio interrompido. Sua mensagem foi preservada para tentar novamente.'
          : failure instanceof Error
            ? failure.message
            : 'Não foi possível enviar a mensagem.',
      );
      return false;
    } finally {
      clearTimeout(timeout);
      controller.current = null;
      setPending(false);
    }
  }
  return { messages, pending, error, send, cancel: () => controller.current?.abort() };
}

import { useId, useRef, useState } from 'react';
import { ThumbsDown, ThumbsUp } from 'lucide-react';
import type { ChatResponse, FeedbackReason, FeedbackRequest } from '../../../shared/contracts';
import { api } from '../../lib/api';

const labels: Record<FeedbackReason, string> = {
  incorrect_specialty: 'A especialidade sugerida não parece adequada',
  unclear_answer: 'A resposta ficou confusa',
  not_helpful: 'A resposta não me ajudou',
  clinic_problem: 'As clínicas apresentadas não foram úteis',
  location_problem: 'Problema com localização/distância',
  insurance_problem: 'Problema com convênio',
  too_slow: 'A resposta demorou muito',
  other: 'Outro',
};
export function ResponseFeedback({ response }: { response: ChatResponse }) {
  const id = useId();
  const [state, setState] = useState<'idle' | 'submitting' | 'submitted' | 'error'>('idle');
  const [rating, setRating] = useState<'positive' | 'negative' | null>(null);
  const [reason, setReason] = useState<FeedbackReason | null>(null);
  const [comment, setComment] = useState('');
  const [panel, setPanel] = useState(false);
  const busy = useRef(false);
  if (!response.feedback) return null;
  const { token, reasons } = response.feedback;
  async function submit(next: FeedbackRequest['rating']) {
    if (busy.current || (next === 'negative' && !reason)) return;
    busy.current = true;
    setRating(next);
    setState('submitting');
    try {
      await api.feedback({
        responseId: response.id,
        token,
        rating: next,
        ...(next === 'negative' && reason ? { reason, comment } : {}),
      });
      setState('submitted');
      setPanel(false);
      setComment('');
    } catch {
      setState('error');
    } finally {
      busy.current = false;
    }
  }
  return (
    <section className="response-feedback" aria-label="Avaliar resposta">
      <p>Essa resposta foi útil?</p>
      <div className="feedback-actions">
        <button
          type="button"
          disabled={state === 'submitting'}
          aria-pressed={rating === 'positive'}
          onClick={() => {
            setPanel(false);
            void submit('positive');
          }}
        >
          <ThumbsUp size={15} /> Sim
        </button>
        <button
          type="button"
          disabled={state === 'submitting'}
          aria-pressed={rating === 'negative'}
          aria-expanded={panel}
          aria-controls={`${id}-panel`}
          onClick={() => {
            setRating('negative');
            setPanel(true);
            setState('idle');
          }}
        >
          <ThumbsDown size={15} /> Não
        </button>
      </div>
      {panel && (
        <form
          id={`${id}-panel`}
          className="feedback-panel"
          onSubmit={(event) => {
            event.preventDefault();
            void submit('negative');
          }}
        >
          <fieldset disabled={state === 'submitting'}>
            <legend>O que poderia melhorar?</legend>
            {reasons.map((value) => (
              <label key={value}>
                <input
                  type="radio"
                  name={`${id}-reason`}
                  value={value}
                  checked={reason === value}
                  onChange={() => setReason(value)}
                  required
                />
                {labels[value]}
              </label>
            ))}
          </fieldset>
          <label htmlFor={`${id}-comment`}>Comentário opcional</label>
          <textarea
            id={`${id}-comment`}
            maxLength={500}
            value={comment}
            onChange={(event) => setComment(event.target.value)}
            disabled={state === 'submitting'}
            aria-describedby={`${id}-privacy`}
            rows={3}
          />
          <small id={`${id}-privacy`}>
            Não inclua sintomas ou dados pessoais. O comentário será salvo para avaliar a utilidade
            do sistema.
          </small>
          <button
            className="primary-button"
            type="submit"
            disabled={!reason || state === 'submitting'}
          >
            Enviar feedback
          </button>
          <button
            className="text-button"
            type="button"
            disabled={state === 'submitting'}
            onClick={() => {
              setPanel(false);
              setComment('');
            }}
          >
            Cancelar
          </button>
        </form>
      )}
      {state === 'submitting' && <p role="status">Enviando feedback…</p>}
      {state === 'submitted' && (
        <p role="status">Obrigado pelo feedback. Você pode alterar seu voto.</p>
      )}
      {state === 'error' && (
        <p role="alert">
          Não foi possível salvar o feedback. Tente novamente; a conversa continua disponível.
        </p>
      )}
    </section>
  );
}

  if (previousLocation !== props.geolocation.location) {
    setPreviousLocation(props.geolocation.location);
    setPage(1);
  }
  const query = JSON.stringify({
    city: props.city,
    insurance: props.insurance,
    specialty,
    q: search,
    page,
    location: searchLocation(props.geolocation.location),
  });
  const key = `${query}|${attempt}`;
  const [result, setResult] = useState<{ key: string; data?: ClinicResults; error?: string }>();
  const loading = result?.key !== key;
  useEffect(() => {
    const controller = new AbortController();
    api
      .clinics(query, controller.signal)
      .then((data) => {
        if (!controller.signal.aborted) setResult({ key, data });
      })
      .catch((error: unknown) => {
        if (!controller.signal.aborted)
          setResult({
            key,
            error:
              error instanceof Error ? error.message : 'Não foi possível consultar o catálogo.',
          });
      });
    return () => controller.abort();
  }, [query, key]);
  return (
    <section className="directory" aria-label="Catálogo de clínicas">
      {props.catalog && (
        <div className="catalog-context">
          <p>
            Dados com fonte e data de consulta. Convênios podem variar por profissional e modalidade
            de atendimento.
          </p>
          <p>
            {props.catalog.directory.realRecords} unidades com fontes públicas em{' '}
            {props.catalog.cities.length} cidades. Cadastro consultado nos sites; sem consulta de
            agenda em tempo real.
          </p>
        </div>
      )}
      <form
        className="directory-search"
        onSubmit={(event) => {
          event.preventDefault();
          setPage(1);
          setSearch(searchInput.trim());
        }}
      >
        <label htmlFor="clinic-search">Nome, bairro ou endereço</label>
        <div>
          <input
            id="clinic-search"
            type="search"
            value={searchInput}
            maxLength={100}
            placeholder="Ex.: Clínica FAM ou Aldeota"
            onChange={(event) => setSearchInput(event.target.value)}
          />
          <button type="submit" className="primary-button">
            <Search size={17} /> Buscar
          </button>
        </div>
      </form>
      <div className="directory-filters">
        <CareFilters
          {...props}
          onCityChange={(value) => {
            setPage(1);
            props.onCityChange(value);
          }}
          onInsuranceChange={(value) => {
            setPage(1);
            props.onInsuranceChange(value);
          }}
        />
        <div className="filter-control specialty-filter">
          <Search size={16} />
          <label htmlFor="specialty">Especialidade</label>
          <select
            id="specialty"
            value={specialty}
            onChange={(event) => {
              setPage(1);
              setSpecialty(event.target.value);
            }}
          >
            <option value="">Todas as especialidades</option>
            {(props.catalog?.specialties ?? []).map(({ id, name }) => (
              <option key={id} value={id}>
                {name}
              </option>
            ))}
          </select>
        </div>
      </div>
      <LocationControl geolocation={props.geolocation} />
      {loading ? (
        <div className="directory-status" role="status">
          <LoaderCircle size={24} className="spin" />
          <p>Consultando o catálogo…</p>
        </div>
      ) : result?.error ? (
        <div className="directory-status" role="alert">
          <p>{result.error}</p>
          <button
            className="primary-button"
            type="button"
            onClick={() => setAttempt((value) => value + 1)}
          >
            Tentar novamente
          </button>
        </div>
      ) : (
        <>
          <div className="directory-count">
            <h2>
              {result?.data?.total ?? 0}{' '}
              {(result?.data?.total ?? 0) === 1 ? 'opção encontrada' : 'opções encontradas'}
            </h2>
            <span>
              {result?.data?.order === 'proximity'
                ? 'Ordenado por proximidade · distância em linha reta'
                : 'Ordenação por nome · sem ranking médico'}
            </span>
          </div>
          {!result?.data?.items.length ? (
            <div className="directory-status">
              <Building2 size={35} />
              <h3>Nenhuma opção para esses filtros</h3>
              <p>Revise a busca ou escolha outra especialidade, cidade ou convênio.</p>
              <button
                type="button"
                className="text-button"
                onClick={() => {
                  setSpecialty('');
                  setSearch('');
                  setSearchInput('');
                  setPage(1);
                  props.onCityChange('');
                  props.onInsuranceChange('');
                }}
              >
                Limpar filtros
              </button>
            </div>
          ) : (
            <div className="clinic-grid">
              {result.data.items.map((clinic) => (
                <ClinicCard
                  key={clinic.id}
                  clinic={clinic}
                  showDistance={result.data?.order === 'proximity'}
                />
              ))}
            </div>
          )}
          {result?.data && result.data.total > result.data.pageSize && (
            <div className="pagination">
              <button
                type="button"
                onClick={() => setPage((value) => value - 1)}
                disabled={page === 1}
                aria-label="Página anterior"
              >
                <ChevronLeft size={18} />
              </button>
              <span>
                Página {page} de {Math.ceil(result.data.total / result.data.pageSize)}
              </span>
              <button
                type="button"
                onClick={() => setPage((value) => value + 1)}
                disabled={page * result.data.pageSize >= result.data.total}
                aria-label="Próxima página"
              >
                <ChevronRight size={18} />
              </button>
            </div>
          )}
        </>
      )}
    </section>
  );
}

import { useEffect, useState } from 'react';
import type { CatalogMeta } from '../../shared/contracts';
import { api } from '../lib/api';

export function useCatalog() {
  const [attempt, setAttempt] = useState(0);
  const [data, setData] = useState<CatalogMeta>();
  const [error, setError] = useState(false);
  useEffect(() => {
    const controller = new AbortController();
    api
      .catalog(controller.signal)
      .then((result) => {
        setData(result);
        setError(false);
      })
      .catch(() => {
        if (!controller.signal.aborted) setError(true);
      });
    return () => controller.abort();
  }, [attempt]);
  return {
    data,
    error,
    retry: () => {
      setError(false);
      setAttempt((value) => value + 1);
    },
  };
}

# API HTTP — versão 2.3

Base local: `http://localhost:3001/api`. JSON UTF-8. Respostas da API usam `Cache-Control: no-store`. Contratos completos: `shared/contracts.ts`.

| Método | Caminho           | Função                                                                           |
| ------ | ----------------- | -------------------------------------------------------------------------------- |
| GET    | `/health`         | Processo ativo                                                                   |
| GET    | `/ready`          | Banco, taxonomia ativa e catálogo disponíveis; RAG opcional                      |
| GET    | `/catalog`        | Especialidades dinâmicas, cidades, convênios, modo e estado do índice textual    |
| GET    | `/clinics`        | Busca compatível sem localização                                                 |
| POST   | `/clinics/search` | Busca com filtros e posição opcional somente no corpo                            |
| POST   | `/chat`           | Navegação, especialidades validadas, clínicas, trechos e comprovante de feedback |
| POST   | `/feedback`       | Gravar/alterar avaliação de uma resposta                                         |

## Catálogo

`GET /catalog` retorna `specialties: [{"id":"urologia","name":"Urologia"}]`, além de `cities`, `insurances`, `mode`, `requiresConsent`, `chatTimeoutMs`, `demoData`, `directory` e `knowledge`. Este último campo foi mantido como metadado de compatibilidade: agora conta **documentos de clínicas**, não fontes médicas para escolher especialidade.

## Busca de clínicas

```json
{
  "city": "",
  "insurance": "Plano Exemplo A",
  "specialty": "urologia",
  "q": "",
  "page": 1,
  "location": { "latitude": -3.7319, "longitude": -38.5267 }
}
```

Envie para `POST /clinics/search`. Todos os filtros são opcionais; cidade/convênio até 80 caracteres, termo até 100 e página 1–1000. `specialty` é um ID de formato slug; a busca só encontra vínculos com especialidades ativas. `location`, se presente, exige dois números finitos: latitude -90..90 e longitude -180..180. Strings numéricas, pares incompletos, campos extras e coordenadas inválidas são rejeitados.

```json
{ "items": [], "total": 0, "page": 1, "pageSize": 6, "order": "proximity" }
```

Cada item mantém os campos de `Clinic`, com `specialtyDetails: [{id,name}]`, `accessNote: string|null` para regras de acesso publicadas e `distanceKm: number|null` computada. `order` é `name` ou `proximity`. Ordenação precede paginação sobre o conjunto completo filtrado. Cidade vazia permite todas as cidades; cidade selecionada é exata, ignorando acentos/caixa. Convênio também é exato. Não há substituições automáticas.

`GET /clinics?city=Fortaleza&insurance=CASSI&specialty=ortopedia&page=1` continua disponível, mas rejeita parâmetros de coordenadas. Não envie posição em URLs.

## Chat

```json
{
  "message": "Estou sentindo ardência ao urinar",
  "history": [],
  "city": "",
  "insurance": "",
  "consent": false,
  "location": { "latitude": -3.7319, "longitude": -38.5267 }
}
```

`message`: 1–2000 caracteres após trim; histórico: até seis mensagens do usuário, 1–2000 cada. A posição é opcional e só vai para o catálogo. `consent=true` é exigido antes de enviar textos à OpenAI. Respostas de segurança e consultas estruturadas ao catálogo funcionam sem consentimento; nesse caso o RAG com embeddings externos também é pulado. O gerador recebe mensagem/histórico redigidos, as perguntas anteriores reconstruídas pelo backend a lista dinâmica `{id,name}` de especialidades ativas e `reportedPatient`, um perfil derivado com parentesco, idade em anos quando identificada e necessidade de confirmar idade. O perfil não contém nome, nascimento ou posição e não é persistido.

Para continuidade, devolva opcionalmente o `context` recebido na resposta anterior:

```json
{
  "message": "e qual dessas fica mais perto?",
  "context": {
    "specialtyId": "oftalmologia",
    "clinicIds": ["public-cldo-aldeota"],
    "city": "",
    "insurance": ""
  },
  "location": { "latitude": -3.7319, "longitude": -38.5267 }
}
```

Use IDs efetivamente retornados pelo servidor; o exemplo ilustra o formato. Contexto não contém relato nem posição, é limitado a seis IDs e fica apenas na memória da aba/requisição. O backend revalida especialidade ativa, visibilidade e filtros das clínicas. Mudança manual dos filtros prevalece sobre o contexto anterior; “dessas” restringe aos IDs anteriores, enquanto “e que aceite Unimed?” refina a especialidade.

A resposta inclui:

- `id`: UUID emitido pelo servidor; `answer`, `urgency`, `questions`, `mode`, `notice`.
- `specialty`: **um** objeto canônico `{id,name,reason?}` ou `null`. Não há array de sugestões nem prioridades. A lista de serviços de cada clínica continua sendo uma lista.
- `intent`: `care_navigation`, `clinic_search` ou `clinic_information`; `mode=catalog` identifica consultas diretas ao catálogo, sem geração médica.
- `context`: `{specialtyId,clinicIds,city,insurance,care?}` ou `null`, para a próxima mensagem. `care` contém somente metadados dos esclarecimentos, descritos abaixo.
- `status` da resposta: `ok`, `llm_unavailable`, `invalid_output`, `needs_information`, `specialty_unrecognized`, `specialty_not_registered` ou `no_matching_clinics`.
- `clinics`: até três resultados do catálogo; `sources`: trechos textuais associados a seus `clinicId` e, quando aplicável, fontes editoriais de explicação/segurança sem `clinicId`. Essas referências gerais não integram o RAG de clínicas.
- `ragStatus`: `skipped`, `not_indexed`, `available`, `no_matches` ou `unavailable`.
- `locationUsed`: booleano, sem devolver a posição enviada.
- `feedback`: `{token,reasons}` ou `null` quando emissão indisponível. `reasons` contém apenas os motivos aplicáveis àquela resposta.

O LLM retorna `{answer,urgency,specialty:{name,reason}|null,questions}`. O contrato antigo de múltiplas sugestões é rejeitado pelo schema estrito. A aplicação resolve a candidata no banco e publica textos controlados pelo backend; nenhum texto livre do modelo chega à interface. Atualize backend e frontend juntos. `routine` não significa ausência de urgência. Fallbacks de orientação usam HTTP 200 com estado explícito; erros de transporte/banco são HTTP 4xx/5xx.

## Esclarecimento de relato genérico

Quando falta contexto, `status=needs_information`, `specialty=null`, `clinics=[]` e `ragStatus=skipped`. `questions` da resposta pública contém até duas perguntas em PT-BR. Reenvie `context` completo com a próxima resposta; o hook da interface já faz isso.

```json
{
  "specialtyId": null,
  "clinicIds": [],
  "city": "",
  "insurance": "",
  "care": {
    "questions": ["onset_duration", "pattern"],
    "askedTopics": ["onset_duration", "pattern"],
    "round": 1
  }
}
```

`care.questions` e `care.askedTopics` são códigos, não dados clínicos. Aceitam `location`, `onset_duration`, `pattern`, `associated_symptoms`, `trigger`, `impact` e `age_group`. Limites: duas perguntas atuais, sete tópicos distintos possíveis, rodada de 1 a 3. O backend reavalia os textos: adulterar o contexto não autoriza uma recomendação. Os metadados não são persistidos e deixam de acompanhar o contexto após a recomendação.

No **contrato interno do LLM**, `questions` passou a aceitar somente esses códigos. Perguntas livres são rejeitadas; no contrato público continuam sendo textos apresentados pelo backend. Não é necessário mudar o modelo instalado: os adaptadores enviam o schema atualizado em cada chamada. Atualize frontend e backend juntos.

Relatos detalhados com uma candidata válida não são bloqueados por perguntas opcionais ou pela incerteza sobre a causa do sintoma. A urgência `uncertain` é preservada quando retornada pelo modelo; não equivale a um diagnóstico nem é convertida em garantia de segurança. Uma candidata ausente continua impedindo recomendação automática.

Quando a idade pode mudar o encaminhamento, `care.questions` inclui `age_group`. Aceita-se resposta de um caractere, como `5`; o histórico permite vinculá-la à pessoa atendida. A idade do responsável não deve ser aplicada à criança. Uma descrição detalhada pode precisar apenas da idade. Pediatria está ativa pela migração 10.

Queixa no ouvido após introdução de um objeto pontiagudo aciona avaliação presencial no mesmo dia: sem perguntas, clínicas eletivas ou RAG. Pode mostrar a área relacionada Otorrinolaringologia, somente se estiver ativa, sem adiar o atendimento. Emergências detectadas anteriormente continuam tendo prioridade.

## Feedback

```json
{
  "responseId": "UUID retornado pelo chat",
  "token": "token retornado em feedback.token",
  "rating": "negative",
  "reason": "unclear_answer",
  "comment": "A explicação poderia ser mais direta."
}
```

Retorno: `{"status":"saved"}`. `rating` aceita `positive` ou `negative`. Positivo não inclui motivo/comentário; negativo exige um motivo permitido pelo comprovante da resposta. Comentário opcional é trimado, limitado a 500 caracteres e redigido antes de salvar. O token é necessário para impedir votos por IDs arbitrários. Não envie modo, versão do prompt, relato, histórico ou coordenadas: campos extras são rejeitados.

Motivos possíveis: `incorrect_specialty`, `unclear_answer`, `not_helpful`, `clinic_problem`, `location_problem`, `insurance_problem`, `too_slow`, `other`. O mesmo `responseId` atualiza o voto por UPSERT. O comprovante expira em sete dias. Não há endpoint público de listagem.

## Controles HTTP

400: dados/consentimento/comprovante inválidos. 403: origem POST não autorizada. 404: rota ausente. 413: corpo acima de 24 KB. 429: limite de uso. 500: falha interna sem expor conteúdo do erro. Erros usam `{"error":"mensagem"}`.

Limite global: `HTTP_RATE_LIMIT` (padrão 120/minuto/IP). O teste de navegador configura 1000 porque executa dezenas de jornadas usando o mesmo IP; o padrão da aplicação permanece 120. Chat: `CHAT_RATE_LIMIT` (padrão 12); feedback: `FEEDBACK_RATE_LIMIT` (padrão 20). Em desenvolvimento, `localhost`, `127.0.0.1` e `[::1]` são aceitos para o mesmo protocolo/porta configurados. Produção exige a origem HTTPS exata. CORS não substitui autenticação ou controle de abuso.

# Arquitetura MedFinder 2.3

## Responsabilidades

1. **Guardrails:** `detectUrgency` avalia relato e histórico antes da IA. Um alerta determinístico não pode ser reduzido pelo modelo, pela localização ou por feedback. Emergência e avaliação no mesmo dia não listam clínicas eletivas.
2. **Intenção e LLM:** após guardrails, `chat-intent.ts` identifica sintomas, pesquisas de clínicas e pedidos de informação. Nomes/aliases de especialidades, clínicas, cidades e convênios vêm do banco. Consultas ao catálogo dispensam geração. Para sintomas, o LLM recebe mensagem/histórico redigidos, perguntas anteriores controladas e a taxonomia ativa; sugere uma candidata ou `null`. Não recebe clínicas, coordenadas ou feedback. JSON Schema e Zod controlam a estrutura.
3. **SpecialtyResolver:** resolve nomes e aliases normalizados por acento/caixa usando `SpecialtyRepository`. Não usa similaridade vetorial nem fuzzy matching. A taxonomia é administrativa, no banco.
4. **CatalogRepository:** fonte de verdade de clínicas, convênios, cidades, endereços, serviços cadastrados e coordenadas. SQL parametrizado filtra atividade, especialidade ativa, convênio exato, cidade exata e palavras opcionais. Não recebe texto arbitrário da IA como SQL.
5. **Proximidade:** Haversine calcula quilômetros em linha reta somente para registros com coordenadas. Ordena todos os candidatos antes de paginar. Sem coordenadas ficam por último; desempate estável por nome e ID. Sem localização, cadastros reais primeiro e nome/ID.
6. **RAG de clínicas:** complementa a resposta com trechos e fontes dos candidatos já selecionados. Não escolhe a especialidade nem amplia filtros. Os trechos são exibidos separadamente, sem uma segunda síntese clínica pelo modelo.
7. **FeedbackRepository:** emite comprovantes e grava avaliações. Não lê nem grava conversas. Não se conecta à IA ou altera os guardrails.

## Especialidades

`specialties` armazena ID, nome, nome normalizado, aliases, atividade e datas. `clinic_specialties` mantém a relação com integridade referencial. A API monta `{id,name}` diretamente do banco; filtros e cards não importam uma lista fixa do frontend.

`specialty_reference` é um vocabulário administrativo separado para reconhecer um nome válido ainda não habilitado. Inicialmente contém a taxonomia cadastrada e Pediatria, Geriatria e Radiologia. A migração 10 promove Pediatria para a taxonomia ativa. Referência de nomenclatura: [CFM, Resolução 2.380/2024](https://sistemas.cfm.org.br/normas/visualizar/resolucoes/BR/2024/2380). Não se trata de uma lista exaustiva nem de validação automática da pertinência clínica da indicação. Psicologia é uma área de cuidado preservada do catálogo anterior.

Resolução ativa retorna `resolved`. Nome só na referência, ou cadastro inativo, retorna `not_registered`. Nome sem correspondência exata/alias explícito retorna `unrecognized`: o LLM não pode declarar sozinho que uma especialidade desconhecida é válida. O comando administrativo `data:import -- specialties` adiciona/atualiza nome, aliases, referência e atividade; conflitos entre nomes/aliases são rejeitados em transação.

Apenas **uma especialidade canônica ou nenhuma** determina a busca de cuidado. A migração 8 acrescenta aliases explícitos, incluindo “oculista”, “oftalmo”, “dermato”, “neuro”, “cardio” e “uro”. Não cria especialidades por saída de IA e não reativa cadastros. Um rótulo desconhecido, inativo ou insuficiente produz fallback sem clínicas, sem o nome original e sem alternativas.

A defesa tem quatro camadas: taxonomia ativa fornecida ao modelo, Structured Output singular, resolução exata no banco e apresentação controlada. `answer` e `reason` gerados continuam sujeitos aos guardrails, mas **não são publicados**; `questions` aceita apenas códigos de tópicos: o backend apresenta mensagens próprias e a entidade resolvida. Isso impede que um alias ou uma especialidade inventada escape pelo texto livre, mesmo quando a candidata principal é válida.

A migração 9 desativa `medicina-de-familia`. O registro e as relações históricas permanecem no banco, mas a área sai da lista fornecida ao modelo, da resolução ativa, dos filtros e dos cards. O snapshot e o texto público da Clínica FAM mantêm somente Ortopedia e Dermatologia dentro do escopo do produto. A clínica permanece no catálogo. Isso é uma escolha de escopo do MedFinder, não uma afirmação sobre o reconhecimento profissional dessa especialidade.

## Esclarecimento condicional

`server/domain/clarification.ts` reconhece relatos claramente genéricos, como dor de cabeça/pescoço sem características, e identifica dados já fornecidos. Isso não é um formulário obrigatório: duração e todas as outras categorias não precisam estar preenchidas. Características, circunstâncias, sintomas associados ou uma descrição mais rica podem permitir uma orientação. Descrições não cobertas pelos padrões linguísticos ficam a cargo do modelo. Trata-se de heurística de linguagem, não validação clínica.

Um relato não genérico com candidata válida segue para resolução no banco. Perguntas opcionais e `urgency=uncertain` não anulam essa candidata; a incerteza é preservada na resposta. Se não houver candidata, não se inventa uma especialidade ou uma nova lista de perguntas para um relato detalhado. Relatos genéricos e idade necessária podem gerar até duas perguntas por rodada, com limite de três rodadas.

`patient.ts` identifica a idade de quem apresenta o sintoma, separando-a da idade do responsável. Filha/filho não implica menor de idade. Idade desconhecida numa possível criança é perguntada antes de recomendar; anos, meses, números por extenso e uma resposta curta como “5” são aceitos. Duração de sintomas não é idade. Nova pessoa reinicia o contexto. A API e o frontend aceitam mensagens a partir de um caractere não vazio.

O perfil derivado (`relationship`, `ageYears`, `adultReported`, `needsAge`) vai ao modelo junto do relato redigido; não contém nome ou nascimento. O prompt instrui a considerar Pediatria para a avaliação inicial de queixas inespecíficas de crianças/adolescentes, sem encaminhar toda criança obrigatoriamente para essa área. Uma filha explicitamente adulta ou com 20 anos ou mais não recebe Pediatria neste MVP. Esse limite conservador do produto não substitui a política etária de cada serviço. Faixa etária efetivamente atendida deve ser confirmada com a unidade.

`care.questions`, `care.askedTopics` e `care.round` ficam em memória. O histórico recente e as perguntas anteriores acompanham a resposta seguinte. A idade também é reconstruída a partir das mensagens, sem tabela de pacientes ou persistência da conversa. Guardrails continuam examinando todo o histórico antes do recorte de contexto. Após três rodadas sem conseguir orientar, não se força indicação.

## Explicação e prioridade presencial

`care-explanation.ts` apresenta uma descrição breve controlada, vinculada ao relato e à área validada. Ouvido e coceira na pele têm descrições específicas de possibilidades, com referências públicas; outras situações usam uma explicação geral de investigação do desconforto. Não são diagnóstico, prescrição ou resumo clínico obtido do RAG. `answer` e `reason` livres do LLM continuam sem ser publicados.

Fontes editoriais de orientação aparecem em `sources` sem `clinicId` e se somam às fontes cadastrais recuperadas pelo RAG. `ragStatus` continua descrevendo apenas o índice de clínicas. Referências de orientação não são prova de validação médica do motor de indicação.

O caso reproduzido da imagem — desconforto no ouvido após uso de lápis — aciona um sinal conservador de possível lesão: avaliação presencial no mesmo dia, sem perguntas, geração ou clínicas eletivas. A área relacionada Otorrinolaringologia aparece apenas se estiver ativa no banco. Alertas de emergência já detectados continuam prevalecendo; não se reduz uma emergência para esse fluxo.

Fontes consultadas em 18/10/2026: [NHS — Earache](https://www.nhs.uk/symptoms/earache/), [NHS — Perforated eardrum](https://www.nhs.uk/conditions/perforated-eardrum/), [NHS — Itchy skin](https://www.nhs.uk/symptoms/itchy-skin/) e [SBP — Pediatria e adolescência](https://www.sbp.com.br/sbp-destaca-papel-do-pediatra-nos-cuidados-com-os-adolescentes/). Não foi criado um protocolo clínico validado.

## Contrato e falhas

O modelo retorna `answer`, `urgency`, `specialty:{name,reason}|null` e `questions`. Limites: zero ou uma candidata, dois tópicos de pergunta, resposta até 1800 caracteres e justificativa até 240. Campos extras e o antigo array são rejeitados. O prompt tem versão `care-navigation-2.3.0`.

`care_navigation` cobre relatos; `clinic_search` consulta especialidade/convênio/cidade/proximidade; `clinic_information` busca fatos de uma unidade. Não há dedução de endereço, contato, plano, horário, serviço ou distância pela memória do LLM. Quando faltam convênio ou unidade identificável, pede esclarecimento. Múltiplas especialidades na mesma consulta também pedem uma escolha.

O frontend devolve contexto curto `{specialtyId,clinicIds,city,insurance}` para continuidades como “dessas” ou “e que aceite Unimed?”. O backend trata-o como não confiável: valida IDs, atividade e filtros novamente. Nada é persistido, nem incluído na entrada do LLM. A história continua limitada a seis mensagens do usuário. Segurança é avaliada antes de toda intenção. Consultas OpenAI sem consentimento usam somente o catálogo e pulam embeddings externos.

`status` distingue `llm_unavailable`, `invalid_output`, `needs_information`, `specialty_unrecognized`, `specialty_not_registered` e `no_matching_clinics`. `ragStatus` é separado: `skipped`, `not_indexed`, `available`, `no_matches`, `unavailable`. Uma falha de embedding não descarta especialidade e clínicas já encontradas. Falhas de banco são erros HTTP de serviço; não são rotuladas como falha da IA.

O modo demo usa casos sintéticos fixos para testar navegação, não inferência clínica. Ollama e OpenAI usam o mesmo contrato. Ollama verifica modelo local antes de enviar texto; OpenAI usa Responses, Structured Outputs e `store:false`. Nenhum erro bruto de SDK é registrado.

## Localização e minimização

`useGeolocation` pertence ao App e é compartilhado entre chat e diretório. Só chama `getCurrentPosition` após clique, com baixa precisão, timeout de 10 segundos e cache do navegador de até 5 minutos. Negação não dispara nova solicitação automática. Cancelar/remover invalida callbacks pendentes.

O frontend envia apenas latitude/longitude arredondadas a quatro casas, em corpos POST; `accuracy` fica no hook. O servidor valida valores finitos e limites. `AIProvider.generate` recebe uma allowlist explícita de campos; o Retriever recebe consulta textual e IDs de clínicas, sem posição do usuário. `locationUsed` é um booleano da resposta, sem guardar coordenadas.

As coordenadas da clínica são dados administrativos persistentes; a posição do usuário e `distanceKm` calculada nunca são persistidas. O bug de distância indisponível vinha do snapshot com **20 pares nulos**, não da permissão do navegador: Haversine corretamente não calculava sem o segundo ponto. O snapshot 2.1 tem **36 pares verificados por unidade**, com auditoria em `data/coordinates.provenance.json`. Importação preserva coordenadas administrativas mais recentes. Não há geocodificação durante o chat. Cidade vazia permite cidades vizinhas; cidade selecionada permanece um filtro exato. Geolocalização não determina urgência, tempo de viagem ou qualidade médica.

## Feedback e proteção contra falsificação

Cada resposta tem UUID aleatório e um token aleatório de 256 bits. Só o hash do token é persistido em `response_receipts`, junto do modo, versão do prompt, motivos permitidos e data. O navegador mantém o token apenas na memória da conversa. O servidor não aceita modo/versão/motivos fora do comprovante.

`POST /api/feedback` faz UPSERT por `response_id`. Negativo exige motivo contextual; comentário opcional tem até 500 caracteres e passa por `redactIdentifiers`. Positivo limpa motivo/comentário anteriores. Não há endpoint público de leitura das avaliações. Comprovantes expiram em 7 dias; a CLI de limpeza remove os vencidos sem apagar os votos existentes. Configure uma política administrativa de retenção do feedback conforme o uso real.

A conversa não depende do feedback: erro na emissão ou no envio não impede o chat. Não há treinamento, fine-tuning, mudança de prompt ou uso de votos no próximo contexto. A redação de identificadores é limitada e não impede que o usuário inclua voluntariamente informações sensíveis num comentário; a UI orienta a evitar isso.

## Limites de escala e validação

O catálogo do MVP é pequeno: distância é calculada em memória sobre todos os registros já filtrados no SQL, depois ordenação e paginação. Antes de ampliar muito o volume, medir memória/latência e adaptar a estratégia. Busca vetorial é exata e restrita por modelo/dimensão em CTE materializada. Veja [RAG.md](RAG.md).

Rate limits são por processo, não uma autorização de identidade. O acesso ao CLI/banco é administrativo. Proxy/APM não devem capturar corpos de saúde, comentários ou localização. Prontidão não comprova conectividade do modelo, revisão clínica ou qualidade dos cadastros; `integrations:check` faz teste técnico sintético explícito.
